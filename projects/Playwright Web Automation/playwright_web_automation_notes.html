01Setup & browser management

A browser context is an isolated session; a page represents a tab.

pip install playwright
playwright install
pip install pytest pytest-playwright   # optional pytest integration

from playwright.sync_api import sync_playwright, expect

with sync_playwright() as p:
    browser = p.chromium.launch(headless=False)
    context = browser.new_context()
    page = context.new_page()
    # ... automation steps ...
    browser.close()

07Waits & synchronization

Playwright auto-waits for actionability. Use explicit waits for a particular state or event.

page.locator("#username").wait_for(state="visible")
page.locator("#loader").wait_for(state="hidden")
page.wait_for_url("**/dashboard")
page.wait_for_load_state("domcontentloaded")

with page.expect_response("**/api/users") as info:
    page.get_by_role("button", name="Load users").click()
response = info.value
print(response.status)
Avoid fixed time.sleep() as your normal wait strategy. Use auto-waiting, web-first assertions or event-based waits.

14Assertions & validations

Web-first assertions retry until the expected condition is met or times out.

from playwright.sync_api import expect

expect(page).to_have_title("Dashboard")
expect(page).to_have_url("https://example.com/dashboard")
expect(page.locator("#welcome")).to_be_visible()
expect(page.locator("#loader")).to_be_hidden()
expect(page.locator("#message")).to_have_text("Login successful")
expect(page.locator("#message")).to_contain_text("successful")
expect(page.locator("#username")).to_have_value("admin")
expect(page.locator("#submit")).to_be_enabled()
expect(page.locator("#agree")).to_be_checked()
expect(page.locator(".row")).to_have_count(5)

))

print(response.json())

18Page Object Model (POM)

Keep page locators and page actions in page classes; keep test validation in tests.

# pages/login_page.py
class LoginPage:
    def __init__(self, page):
        self.page = page
        self.username = page.get_by_label("Username")
        self.password = page.get_by_label("Password")
        self.login_button = page.get_by_role("button", name="Login")

    def open(self):
        self.page.goto("https://example.com/login")

    def login(self, username, password):
        self.username.fill(username)
        self.password.fill(password)
        self.login_button.click()

# test_login.py
from playwright.sync_api import expect
from pages.login_page import LoginPage

def test_login(page):
    login = LoginPage(page)
    login.open()
    login.login("test-user", "test-password")
    expect(page.get_by_text("Welcome")).to_be_visible()

20Complete login automation example

A single script combining browser setup, navigation, locators, actions and validation.

from playwright.sync_api import sync_playwright, expect

with sync_playwright() as p:
    browser = p.chromium.launch(headless=False)
    context = browser.new_context()
    page = context.new_page()

    page.goto("https://example.com/login")
    page.get_by_label("Username").fill("test-user")
    page.get_by_label("Password").fill("test-password")
    page.get_by_role("button", name="Login").click()

    expect(page.get_by_text("Welcome")).to_be_visible()
    print("Page title:", page.title())

    browser.close()
Learning path: browser setup → locators → interactions → waits → assertions → tabs/frames/files → network → pytest → POM → debugging. Use automation only on websites and systems you are authorized to test.
